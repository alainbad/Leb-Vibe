## Triggers

### 1. Create profile on auth signup
```sql
CREATE OR REPLACE FUNCTION handle_new_user()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO public.profiles (id, full_name, avatar_url)
  VALUES (
    NEW.id,
    NEW.raw_user_meta_data->>'full_name',
    NEW.raw_user_meta_data->>'avatar_url'
  );
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION handle_new_user();
```

### 2. Update venue search_vector
```sql
CREATE OR REPLACE FUNCTION update_venue_search_vector()
RETURNS TRIGGER AS $$
BEGIN
  NEW.search_vector =
    setweight(to_tsvector('english', COALESCE(unaccent(NEW.name), '')), 'A') ||
    setweight(to_tsvector('english', COALESCE(unaccent(NEW.description), '')), 'B') ||
    setweight(to_tsvector('english', COALESCE(unaccent(NEW.address), '')), 'C');
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER venue_search_vector_update
  BEFORE INSERT OR UPDATE OF name, description, address ON venues
  FOR EACH ROW EXECUTE FUNCTION update_venue_search_vector();
```

### 3. Auto-update avg_rating and review_count on review change
```sql
CREATE OR REPLACE FUNCTION update_venue_rating()
RETURNS TRIGGER AS $$
BEGIN
  UPDATE venues SET
    avg_rating = (
      SELECT COALESCE(AVG(rating_overall), 0)
      FROM reviews
      WHERE venue_id = COALESCE(NEW.venue_id, OLD.venue_id) AND status = 'approved'
    ),
    review_count = (
      SELECT COUNT(*)
      FROM reviews
      WHERE venue_id = COALESCE(NEW.venue_id, OLD.venue_id) AND status = 'approved'
    ),
    updated_at = now()
  WHERE id = COALESCE(NEW.venue_id, OLD.venue_id);
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER review_update_venue_rating
  AFTER INSERT OR UPDATE OR DELETE ON reviews
  FOR EACH ROW EXECUTE FUNCTION update_venue_rating();
```

### 4. Auto-manage is_new_opening flag
```sql
CREATE OR REPLACE FUNCTION set_new_opening_flag()
RETURNS TRIGGER AS $$
BEGIN
  NEW.is_new_opening = (NEW.created_at > NOW() - INTERVAL '60 days');
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER venue_set_new_opening
  BEFORE INSERT ON venues
  FOR EACH ROW EXECUTE FUNCTION set_new_opening_flag();
```

Helper function:
```sql
CREATE OR REPLACE FUNCTION get_my_role()
RETURNS user_role AS $$
  SELECT role FROM profiles WHERE id = auth.uid()
$$ LANGUAGE sql STABLE SECURITY DEFINER;

CREATE OR REPLACE FUNCTION is_admin()
RETURNS boolean AS $$
  SELECT role IN ('admin', 'super_admin') FROM profiles WHERE id = auth.uid()
$$ LANGUAGE sql STABLE SECURITY DEFINER;

CREATE OR REPLACE FUNCTION owns_venue(venue_uuid uuid)
RETURNS boolean AS $$
  SELECT EXISTS (
    SELECT 1 FROM venues WHERE id = venue_uuid AND owner_id = auth.uid()
    UNION
    SELECT 1 FROM venue_staff WHERE venue_id = venue_uuid AND user_id = auth.uid() AND is_active = true
  )
$$ LANGUAGE sql STABLE SECURITY DEFINER;
```

### venues policies
```sql
-- Public read for active venues
CREATE POLICY "venues_public_read" ON venues FOR SELECT
  USING (status = 'active');

-- Owners/staff see their own (any status)
CREATE POLICY "venues_owner_read" ON venues FOR SELECT
  USING (owns_venue(id));

-- Admin sees all
CREATE POLICY "venues_admin_all" ON venues FOR ALL
  USING (is_admin());

-- Owners can update (but not change status themselves)
CREATE POLICY "venues_owner_update" ON venues FOR UPDATE
  USING (owner_id = auth.uid())
  WITH CHECK (status = (SELECT status FROM venues WHERE id = id));

-- Authenticated users can insert (creates pending venue)
CREATE POLICY "venues_owner_insert" ON venues FOR INSERT
  WITH CHECK (auth.uid() IS NOT NULL AND owner_id = auth.uid());
```

```sql
  USING (owns_venue(venue_id));

  USING (is_admin());

-- Authenticated users can create

  WITH CHECK (auth.uid() IS NOT NULL AND user_id = auth.uid());

-- Users can cancel own confirmed reservations
CREATE POLICY "reservations_user_cancel" ON reservations FOR UPDATE
  USING (user_id = auth.uid())
  WITH CHECK (status = 'cancelled');

CREATE POLICY "reservations_venue_update" ON reservations FOR UPDATE
  USING (owns_venue(venue_id));
```

```sql
  USING (user_id = auth.uid());

  USING (owns_venue(venue_id));

CREATE POLICY "reviews_admin" ON reviews FOR ALL
  USING (is_admin());

CREATE POLICY "reviews_create" ON reviews FOR INSERT
  WITH CHECK (auth.uid() IS NOT NULL AND user_id = auth.uid());

-- Users can update own pending reviews
CREATE POLICY "reviews_own_update" ON reviews FOR UPDATE
  USING (user_id = auth.uid() AND status = 'pending');

-- Venue can add reply only
CREATE POLICY "reviews_venue_reply" ON reviews FOR UPDATE
  USING (owns_venue(venue_id))
  WITH CHECK (venue_reply IS NOT NULL);
```
